use std::io::{Read, Seek, SeekFrom};

pub const PACKAGE_FILE_TAG: u32 = 0x9E2A83C1;
pub const PACKAGE_FILE_TAG_SWAPPED: u32 = 0xC1832A9E;

#[derive(Debug, Clone)]
pub struct PackageHeader {
    pub magic: u32,
    pub file_version_ue4: u32,
    pub package_flags: u32,
    pub name_count: i32,
    pub name_offset: i32,
    pub gatherable_texture_data_count: i32,
    pub gatherable_texture_data_offset: i32,
    pub export_count: i32,
    pub export_offset: i32,
    pub import_count: i32,
    pub import_offset: i32,
    pub depends_offset: i32,
    pub depends_count: i32,
    pub soft_package_references_count: i32,
    pub soft_package_references_offset: i32,
    pub searchable_names_offset: i32,
    pub searchable_names_count: i32,
    pub thumbnail_table_offset: i32,
    pub guid: [u8; 16],
    pub generations: Vec<GenerationInfo>,
    pub engine_version: u32,
    pub cooked_content: u32,
    pub package_version: u32,
}

#[derive(Debug, Clone)]
pub struct GenerationInfo {
    pub export_count: i32,
    pub name_count: i32,
}

impl PackageHeader {
    pub fn read<R: Read + Seek>(reader: &mut R) -> Result<Self, super::Error> {
        let mut magic = [0u8; 4];
        reader.read_exact(&mut magic)?;
        let magic = u32::from_le_bytes(magic);
        
        if magic != PACKAGE_FILE_TAG && magic != PACKAGE_FILE_TAG_SWAPPED {
            return Err(super::Error::InvalidMagic(magic));
        }

        let _legacy_file_version = read_i32(reader)?;  // -9 for UE4.24+
        let _legacy_ue3_version = read_i32(reader)?;
        let file_version_ue4 = read_u32(reader)?;
        let _file_version_ue5 = read_u32(reader)?;  // 0 for UE4

        let mut guid = [0u8; 16];
        reader.read_exact(&mut guid)?;

        let package_flags = read_u32(reader)?;
        let name_count = read_i32(reader)?;
        let name_offset = read_i32(reader)?;

        if file_version_ue4 >= 342 {
            let gatherable_texture_data_count = read_i32(reader)?;
            let gatherable_texture_data_offset = read_i32(reader)?;
        }
        // ... continues with export, import, depends...
    }
}

### Task 4: Implement FName table parsing

FName entries are serialized as:
- int32: Name length (or index into a name map)
- char[]: Name string (null-terminated? depends on version)
- int32: Flags (non-zero for some names)

For cooked UE4.27, names may use a different encoding (stored as an index into the name map embedded in the executable). But for the uasset files we care about, the non-cooked format stores names inline.

Actually, in cooked uassets (which PW ships with), the name table is different. Let me investigate the actual format by examining real files.
