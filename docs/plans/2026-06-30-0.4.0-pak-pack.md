# Project Modman 0.4.0 — PAK Pack

**Goal:** Implement `modman pack` to create .pak files from directories.

---

### Task 1: Add `pack` function to modman-pak

**Files:**
- Modify: `crates/modman-pak/src/lib.rs`

Add a standalone `pack` function (no archive needs to be opened first):

```rust
/// Create a .pak file from a directory
pub fn pack(
    input_dir: impl AsRef<Path>,
    output_path: impl AsRef<Path>,
    version: Version,
    mount_point: String,
    compression: Option<Compression>,
) -> Result<(), Error> {
    let input_dir = input_dir.as_ref();
    let output_path = output_path.as_ref();

    let mut paths = Vec::new();
    collect_files(&mut paths, input_dir)?;
    paths.sort();

    let compress_vec: Vec<Compression> = compression.into_iter().collect();
    let mut pak = repak::PakBuilder::new()
        .compression(compress_vec.clone())
        .writer(
            std::io::BufWriter::new(std::fs::File::create(output_path)?),
            version,
            mount_point,
            None,
        );

    for path in &paths {
        let relative = path.strip_prefix(input_dir).expect("path under input dir");
        let data = std::fs::read(path)?;
        let path_str = relative.to_string_lossy().replace('\\', "/");
        pak.write_file(&path_str, compression.is_some(), data)?;
    }

    pak.write_index()?;
    Ok(())
}

fn collect_files(paths: &mut Vec<std::path::PathBuf>, dir: &Path) -> std::io::Result<()> {
    for entry in std::fs::read_dir(dir)? {
        let entry = entry?;
        let path = entry.path();
        if path.is_dir() {
            collect_files(paths, &path)?;
        } else {
            paths.push(path);
        }
    }
    Ok(())
}
```

### Task 2: Wire up `pack` command in CLI

**Files:**
- Modify: `crates/modman-cli/src/main.rs`

Update Pack struct:
```rust
/// Pack a directory into a .pak file
Pack {
    /// Input directory
    input: String,
    /// Output path. Defaults to input directory name + .pak
    #[arg(short, long)]
    output: Option<String>,
    /// Mount point
    #[arg(short, long, default_value = "../../../")]
    mount_point: String,
    /// PAK version (default: V11 for UE4.27)
    #[arg(long, default_value = "V11")]
    version: String,
    /// Compression algorithm (zlib, gzip, zstd, lz4)
    #[arg(long)]
    compression: Option<String>,
},
```

Handler:
```rust
Some(Commands::Pack {
    input,
    output,
    mount_point,
    version,
    compression,
}) => {
    let output_path = output.unwrap_or_else(|| format!("{}.pak", input));
    let version = match version.to_uppercase().as_str() {
        "V8B" => modman_pak::Version::V8B,
        "V11" => modman_pak::Version::V11,
        _ => {
            eprintln!("Error: unsupported version '{}'. Use V8B or V11.", version);
            std::process::exit(1);
        }
    };
    let compression = compression.as_deref().and_then(|c| match c.to_lowercase().as_str() {
        "zlib" => Some(modman_pak::Compression::Zlib),
        "gzip" => Some(modman_pak::Compression::Gzip),
        "zstd" => Some(modman_pak::Compression::Zstd),
        "lz4" => Some(modman_pak::Compression::LZ4),
        _ => {
            eprintln!("Error: unsupported compression '{}'. Use zlib, gzip, zstd, or lz4.", c);
            std::process::exit(1);
        }
    });
    
    println!("Packing {} to {} (v{}, mount: {}) ...", input, output_path, version, mount_point);
    match modman_pak::pack(&input, &output_path, version, mount_point, compression) {
        Ok(()) => println!("Done!"),
        Err(e) => {
            eprintln!("Error: {}", e);
            std::process::exit(1);
        }
    }
}
```
